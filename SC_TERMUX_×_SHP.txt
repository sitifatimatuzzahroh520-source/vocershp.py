def read_unique_lines(path):
    if not path.exists():
        return []
    return list(dict.fromkeys(line.strip() for line in path.read_text(encoding="utf-8-sig").splitlines() if line.strip()))


def save_unique_success(path, data_str):
    existing = read_unique_lines(path)
    if data_str not in existing:
        with path.open("a", encoding="utf-8") as f:
            f.write(data_str + "\n")


def get_next_fingerprint():
    """Mengambil baris fingerprint secara bergantian (rotasi) dari file fingerprint.txt"""
    global fp_index
    fps = read_unique_lines(FINGERPRINT_FILE)
    if not fps:
        return None
    selected_fp = fps[fp_index % len(fps)]
    fp_index += 1
    return selected_fp


async def click_button_by_keywords(keywords, after_id=0):
    for _ in range(12):
        messages = await client.get_messages(BOT_USERNAME, limit=12)
        for msg in messages:
            if not msg.out and (after_id == 0 or msg.id > after_id) and msg.buttons:
                for row in msg.buttons:
                    for button in row:
                        label = (button.text or "").lower()
                        if any(k in label for k in keywords):
                            await button.click()
                            return msg.id
        await asyncio.sleep(2)
    return None


async def wait_for_bot_prompt(phrases, after_id=0, timeout=35):
    loop = asyncio.get_running_loop()
    deadline = loop.time() + timeout
    while loop.time() < deadline:
        messages = await client.get_messages(BOT_USERNAME, limit=15)
        for msg in messages:
            if msg.out or (after_id > 0 and msg.id <= after_id):
                continue
            content = (msg.raw_text or "").casefold()
            if any(phrase in content for phrase in phrases):
                return msg.id
        await asyncio.sleep(2)
    return None
